    DROP y DELETE

    DROP y DELETE sirven para eliminar tablas y filas en una base de datos.

    DROP

    DROP elimina tablas y sus filas:

    CREATE TABLE ejemplo (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  dato TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

INSERT INTO ejemplo(dato) VALUES 
('hola'),
('hola'),
('hola'),
('hola');

SELECT * FROM ejemplo;

DROP TABLE ejemplo;

SELECT * FROM ejemplo;

    Al ejecutarlo:

    ❯ sqlite3 test.db < drop.sql
1|hola|2026-09-05 10:05:17
2|hola|2026-09-05 10:05:17
3|hola|2026-09-05 10:05:17
4|hola|2026-09-05 10:05:17
Parse error near line 17: no such table: ejemplo

    Cuando una tabla depende de otra por una relación, debemos eliminar antes la tabla padre.

    Considera el siguiente ejemplo:

    PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS usuarios (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nombre TEXT NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS mensajes (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  content TEXT NOT NULL,
  usuario_id INTEGER,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY(usuario_id) REFERENCES usuarios(id)
);

INSERT INTO usuarios (nombre) VALUES 
('admin'),
('user0');

INSERT INTO mensajes (content, usuario_id)
VALUES 
('hola mundo', (SELECT id FROM usuarios WHERE nombre = 'admin')),
('que tal todos?', (SELECT id FROM usuarios WHERE nombre = 'admin')),
('bien! que tal tu?', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0'));

    Si intentamos borrar la tabla de usuarios obtenemos un error,

    /tmp/test
❯ sqlite3 test.db < schema.sql

/tmp/test
❯ sqlite3 test.db "PRAGMA foreign_keys = ON; DROP TABLE usuarios;"
Error: stepping, FOREIGN KEY constraint failed (19)

    Ya que la tabla mensajes depende de ella, si borramos primero mensajes no hay problema:

    /tmp/test
❯ sqlite3 test.db "PRAGMA foreign_keys = ON; DROP TABLE mensajes;"
/tmp/test
❯ sqlite3 test.db "PRAGMA foreign_keys = ON; DROP TABLE usuarios;"
/tmp/test
❯ sqlite3 test.db ".tables"

    Recuerda que si no activas las foreign_keys SQLite ni se entera:

    /tmp/test
❯ sqlite3 test.db < schema.sql

/tmp/test
❯ sqlite3 test.db "DROP TABLE usuarios;"

/tmp/test
❯ sqlite3 test.db ".tables"
mensajes

    Ahora mensajes tiene una columna que no apunta a ninguna tabla que exista.

    DELETE

    Elimina una fila segun el criterio que necesites:

    CREATE TABLE ejemplo (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  dato TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

INSERT INTO ejemplo(dato) VALUES 
('hola'),
('hola'),
('hola'),
('hola');

DELETE FROM ejemplo WHERE id = 1;
DELETE FROM ejemplo WHERE id = 2;
DELETE FROM ejemplo WHERE id = 3;

    Cuando leas la tabla:

    ❯ sqlite3 test.db "SELECT * FROM ejemplo"
4|hola|2026-09-05 10:29:29

    No podemos borrar el user0 o nuestra tabla de mensajes tendrá inconsistencias, primero debemos borrar sus mensajes:

    /tmp/test
❯ sqlite3 test.db "PRAGMA foreign_keys = ON; DELETE FROM mensajes WHERE usuario_id = 2"

/tmp/test
❯ sqlite3 test.db "PRAGMA foreign_keys = ON; DELETE FROM usuarios WHERE nombre = 'user0'"

    ON DELETE CASCADE

    Podemos hacer que la base de datos haga esto automaticamente cuando declaramos nuestras claves foraneas:

    PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS usuarios (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nombre TEXT NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS mensajes (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  content TEXT NOT NULL,
  usuario_id INTEGER,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY(usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
);

INSERT INTO usuarios (nombre) VALUES 
('admin'),
('user0');

INSERT INTO mensajes (content, usuario_id)
VALUES 
('hola mundo', (SELECT id FROM usuarios WHERE nombre = 'admin')),
('que tal todos?', (SELECT id FROM usuarios WHERE nombre = 'admin')),
('bien! que tal tu?', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0')),
('otro mensaje', (SELECT id FROM usuarios WHERE nombre = 'user0'));

DELETE FROM usuarios WHERE nombre = 'user0';

SELECT * FROM mensajes;

    Tan solo hemos añadido FOREIGN KEY(usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE, pero ahora cuando borramos un usuario no da error, y se borra tanto el usuario como los mensajes asociados al mismo, pero ahora cuando borramos un usuario no da error, y se borra tanto el usuario como los mensajes asociados al mismo.
